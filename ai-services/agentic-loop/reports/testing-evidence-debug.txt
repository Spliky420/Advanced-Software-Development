# Ruff configuration for Le Hoa Long's Python (LeHoaLong/backend, /database
# and /tests).
#
# Scoped to this directory on purpose: ruff walks up from each file it is
# given and uses the nearest config, so this lints my code without imposing a
# style on the other five students' directories. CI runs:
#
#   ruff check LeHoaLong
#
# `ruff format` is deliberately NOT a CI gate. The rules below catch real
# mistakes -- unused imports, undefined names, shadowed builtins, mutable
# default arguments, insecure patterns -- and formatting is left to the author.
#
# The same config runs from the root .pre-commit-config.yaml, so a finding
# stops the commit locally before CI ever sees it.

# Matches python:3.11-slim in the backend and database Dockerfiles, so ruff
# judges the code against the interpreter the containers actually ship.
target-version = "py311"

# Wider than the 88 ruff defaults to. The SQL strings and the assertion
# messages in this feature read worse broken across lines than run long.
line-length = 120

exclude = [
    "frontend",       # JavaScript -- eslint's job, not ruff's.
    ".pytest_cache",
]

[lint]
select = [
    "E",    # pycodestyle errors
    "W",    # pycodestyle warnings
    "F",    # pyflakes -- unused imports, undefined names
    "I",    # isort -- import ordering
    "UP",   # pyupgrade -- syntax that is outdated for target-version
    "B",    # flake8-bugbear -- likely bugs, e.g. mutable default arguments
    "C4",   # flake8-comprehensions
    "S",    # flake8-bandit -- security: SQL built from strings, bind-all, assert, ...
]

[lint.per-file-ignores]
# The tests import `app` only after editing sys.path, so those imports
# cannot come first. conftest.py already carries an inline noqa; this keeps
# a future test file from tripping over the same unavoidable ordering.
#
# S101 (assert): pytest is built on bare asserts, so in tests the rule is
# all noise (813 hits). Outside tests it stays on, because asserts are
# stripped under `python -O` -- a check that must hold is an explicit raise.
"tests/*" = ["E402", "S101"]

"endpoint_tests/*" = ["S101"]

# tools/build_corpus.py holds the corpus prose itself, as string literals.
# A fact wrapped across two source lines is harder to read and easy to break
# with a stray space or a missing one, and the point of that file is that the
# sentences in it are exactly the sentences that ship to the shared corpus.
"tools/build_corpus.py" = ["E501"]

# Cloud deployment to Azure Container Apps.
#
# Runs every student's CI workflow first (each one is called through its
# workflow_call trigger, so it runs in full regardless of which paths
# changed). Only if all six pass does the deploy job run: it logs in to Azure
# with OIDC as a user-assigned managed identity, builds and pushes every image
# to ACR, and updates the container apps -- using the same scripts as a manual
# deploy (scripts/deploy/02-build-images.sh and 03-deploy-apps.sh).
#
# No passwords or client secrets: azure/login exchanges GitHub's OIDC token
# for an Azure token. The three IDs it needs are repository variables, not
# secrets. Infrastructure (scripts/deploy/01-infra.sh) is created once by
# hand; this workflow's identity can only push images and update apps in the
# project's resource group.

name: cloud-deployment

on:
  push:
    branches:
      - main
  workflow_dispatch:

# Default for every job; the deploy job adds id-token: write for OIDC.
permissions:
  contents: read

# Never run two deploys at once; queue instead of cancelling a deploy halfway.
concurrency:
  group: cloud-deployment
  cancel-in-progress: false

jobs:
  # --- Gate: every student's CI must pass ------------------------------------
  joshua-ci:
    uses: ./.github/workflows/joshua-ci.yml
  maxwell-ci:
    uses: ./.github/workflows/maxwell-ci.yml
  thomas-ci:
    uses: ./.github/workflows/thomas-ci.yml
  enerel-ci:
    uses: ./.github/workflows/enerel-ci.yml
  hyunwoo-ci:
    uses: ./.github/workflows/hyunwoo-ci.yml
  lehoalong-ci:
    uses: ./.github/workflows/lehoalong-ci.yml

  # --- Deploy ------------------------------------------------------------------
  deploy:
    needs: [joshua-ci, maxwell-ci, thomas-ci, enerel-ci, hyunwoo-ci, lehoalong-ci]
    runs-on: ubuntu-latest
    permissions:
      contents: read
      id-token: write   # lets azure/login request GitHub's OIDC token
    env:
      IMAGE_TAG: ${{ github.sha }}
      BUILD_MODE: docker   # ACR Tasks are not available on Azure for Students
    steps:
      - uses: actions/checkout@v4

      - name: Azure login (OIDC, user-assigned managed identity)
        uses: azure/login@v2
        with:
          client-id: ${{ vars.AZURE_CLIENT_ID }}
          tenant-id: ${{ vars.AZURE_TENANT_ID }}
          subscription-id: ${{ vars.AZURE_SUBSCRIPTION_ID }}

      - name: Build and push images
        run: bash scripts/deploy/02-build-images.sh

      - name: Update container apps
        run: bash scripts/deploy/03-deploy-apps.sh

      - name: Smoke test
        run: |
          source scripts/deploy/config.sh
          set +e   # report every check, then fail at the end if any failed
          domain="$(env_domain)"
          check() {  # url expected-text; retries while a replica wakes up
            local body
            for i in $(seq 1 12); do
              body="$(curl -fsS --max-time 30 "$1")" && [[ "$body" == *"$2"* ]] && { echo "OK   $1"; return 0; }
              sleep 10
            done
            echo "FAIL $1"; return 1
          }
          status=0
          check "https://shared-frontend.$domain/" "joshua-frontend.$domain" || status=1
          for s in joshua maxwell thomas enerel hyunwoo lehoalong; do
            check "https://$s-frontend.$domain/" "<" || status=1
          done
          check "https://joshua-frontend.$domain/api/holdings" "ticker" || status=1
          # Goals and Budgeting: a CRUD read through the frontend's nginx proxy
          # to the internal backend and its database.
          check "https://lehoalong-frontend.$domain/api/goals" "goal_id" || status=1
          # Research Library: a CRUD read through the frontend's nginx proxy.
          check "https://enerel-frontend.$domain/api/documents" "doc_type" || status=1
          {
            echo "### Deployed \`${IMAGE_TAG::7}\`"
            echo
            echo "- Home page: https://shared-frontend.$domain"
            for s in joshua maxwell thomas enerel hyunwoo lehoalong; do
              echo "- $s: https://$s-frontend.$domain"
            done
          } >> "$GITHUB_STEP_SUMMARY"
          exit $status
